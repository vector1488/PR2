```java
// Ferma.java
import java.util.Scanner;

public class Ferma {
    private static long powerMod(long a, long b, long c) {
        long result = 1;
        a = a % c;
        while (b > 0) {
            if ((b & 1) == 1) result = (result * a) % c;
            a = (a * a) % c;
            b >>= 1;
        }
        return result;
    }

    private static boolean isPrime(long p) {
        if (p < 2) return false;
        if (p % 2 == 0) return p == 2;
        for (long i = 3; i * i <= p; i += 2) {
            if (p % i == 0) return false;
        }
        return true;
    }

    private static boolean checkFerma(long a, long x, long p) {
        if (a % p == 0) return true;
        return powerMod(a, p - 1, p) == 1;
    }

    private static long calculateFerma(long a, long x, long p) {
        if (a % p == 0) return 0;
        x = x % (p - 1);
        return powerMod(a, x, p);
    }

    public static void run() {
        Scanner sc = new Scanner(System.in);
        System.out.println("Введите числа для a^x mod p (p должно быть простым):");
        long a = sc.nextLong(), x = sc.nextLong(), p = sc.nextLong();

        if (!isPrime(p)) {
            System.out.println("Число p не простое. Запустите заново.");
            return;
        }
        if (!checkFerma(a, x, p)) {
            System.out.println("Теорема не выполняется. Запустите заново.");
            return;
        }
        long res = calculateFerma(a, x, p);
        System.out.println("Результат: " + res);
    }
}
```

```java
// Extevklid.java
import java.util.Scanner;

public class Extevklid {
    private static int[] extendedGcd(int a, int b) {
        int x1 = a, y1 = b;
        int x2 = 1, y2 = 0;
        int x3 = 0, y3 = 1;

        while (y1 != 0) {
            int q = x1 / y1;
            int t1 = x1 % y1;
            int t2 = x2 - q * y2;
            int t3 = x3 - q * y3;

            x1 = y1; x2 = y2; x3 = y3;
            y1 = t1; y2 = t2; y3 = t3;
        }
        return new int[]{x1, x2, x3}; // gcd, u, v
    }

    private static int findD(int m, int n) {
        int[] r = extendedGcd(m, n);
        int gcd = r[0], x = r[1];
        if (gcd != 1) throw new RuntimeException("Обратного элемента не существует");
        return (x % n + n) % n;
    }

    public static void run() {
        Scanner sc = new Scanner(System.in);
        System.out.println("Введите c и m (для уравнения c*d mod m = 1):");
        int c = sc.nextInt(), m = sc.nextInt();
        try {
            int d = findD(c, m);
            System.out.println("Обратный элемент d = " + d);
            System.out.println("Проверка: " + (c * d) % m);
        } catch (RuntimeException e) {
            System.out.println("Ошибка: " + e.getMessage());
        }
    }
}
```

```java
// Modules.java
import java.util.Scanner;

public class Modules {
    private static int modInverse(int a, int m) {
        int m0 = m, y = 0, x = 1;
        if (m == 1) return 0;
        if (a == m) return 1;

        while (a > 1) {
            int q = a / m;
            int t = m;
            m = a % m;
            a = t;
            t = y;
            y = x - q * y;
            x = t;
        }
        if (x < 0) x += m0;
        return (a == 1) ? x : -1;
    }

    public static void run() {
        Scanner sc = new Scanner(System.in);
        System.out.println("c^-1 mod m = d");
        System.out.print("Введите значения c и m: ");
        int c = sc.nextInt(), m = sc.nextInt();
        int inv = modInverse(c, m);
        if (inv == -1) {
            System.out.println("Обратного элемента для " + c + " mod " + m + " не существует!");
        } else {
            System.out.println("Обратный элемент = " + inv);
            System.out.println("Проверка: " + (c * inv) % m);
        }
    }
}
```

```java
// Fraction.java
import java.util.*;

public class Fraction {
    private static List<Integer> continuedFraction(int p, int q, int[] m) {
        List<Integer> cf = new ArrayList<>();
        while (q != 0) {
            int a = p / q;
            cf.add(a);
            int r = p % q;
            p = q;
            q = r;
        }
        m[0] = p;
        return cf;
    }

    public static void run() {
        Scanner sc = new Scanner(System.in);
        System.out.println("Введите значения множителей a и b:");
        int p = sc.nextInt(), q = sc.nextInt();
        int[] m = new int[1];
        List<Integer> cf = continuedFraction(p, q, m);
        System.out.print(p + "/" + q + " = [");
        for (int i = 0; i < cf.size(); i++) {
            System.out.print(cf.get(i));
            if (i + 1 < cf.size()) System.out.print(";");
        }
        System.out.println("]");
        System.out.println("НОД: " + m[0]);
    }
}
```

```java
// SSSSen.java
import java.util.*;

public class SSSSen {
    public static final int PRIME = 257;

    public static int modPow(int base, int exp, int mod) {
        long result = 1;
        long b = base % mod;
        long e = exp;
        while (e > 0) {
            if ((e & 1) == 1) result = (result * b) % mod;
            b = (b * b) % mod;
            e >>= 1;
        }
        return (int) result;
    }

    public static List<int[]> shamirSplit(int secret, int n, int k) {
        Random rng = new Random();
        int[] coeffs = new int[k];
        coeffs[0] = secret % PRIME;
        for (int i = 1; i < k; i++) coeffs[i] = 1 + rng.nextInt(PRIME - 1);

        List<int[]> shares = new ArrayList<>();
        for (int x = 1; x <= n; x++) {
            int y = 0;
            int pow = 1;
            for (int i = 0; i < k; i++) {
                y = (y + coeffs[i] * pow) % PRIME;
                pow = (pow * x) % PRIME;
            }
            shares.add(new int[]{x, y});
        }
        return shares;
    }

    public static int shamirCombine(List<int[]> shares) {
        long secret = 0;
        int t = shares.size();
        for (int i = 0; i < t; i++) {
            long xi = shares.get(i)[0];
            long yi = shares.get(i)[1];
            long num = 1, den = 1;
            for (int j = 0; j < t; j++) {
                if (i == j) continue;
                long xj = shares.get(j)[0];
                num = (num * (PRIME - xj)) % PRIME; // -xj mod PRIME
                den = (den * ((xi - xj + PRIME) % PRIME)) % PRIME;
            }
            long inv = modPow((int)den, PRIME - 2, PRIME);
            long term = yi * num % PRIME * inv % PRIME;
            secret = (secret + term) % PRIME;
        }
        return (int) secret;
    }
}
```

```java
// Encrypt.java
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.List;
import java.util.Random;

public class Encrypt {
    private static int generateKey() {
        Random rng = new Random();
        return 1 + rng.nextInt(255);
    }

    public static int encrypt(String inFile) {
        Path in = Path.of(inFile);
        Path out = Path.of("encrypted.bin");
        try {
            byte[] data = Files.readAllBytes(in);
            int key = generateKey();
            System.out.println("Случайный ключ: " + key);

            int n = 5, k = 3;
            List<int[]> shares = SSSSen.shamirSplit(key, n, k);
            System.out.println("Секрет разделен на части (n=" + n + ", k=" + k + "):");
            for (int[] sh : shares) System.out.println(" (" + sh[0] + ", " + sh[1] + ")");

            byte[] outData = new byte[data.length];
            for (int i = 0; i < data.length; i++) outData[i] = (byte) (data[i] ^ (byte) key);

            Files.write(out, outData);
            System.out.println("Файл зашифрован и сохранен в " + out.toString());
            return 0;
        } catch (IOException e) {
            System.err.println("Ошибка: " + e.getMessage());
            return 1;
        }
    }
}
```

```java
import java.util.*;

public class SSSSen {
    // Очень упрощённая версия схемы Шамира (только для демонстрации)

    public static void splitSecret() {
        Scanner sc = new Scanner(System.in);
        System.out.print("Введите секрет (целое число): ");
        int secret = sc.nextInt();
        System.out.print("Введите количество частей: ");
        int n = sc.nextInt();

        Random rand = new Random();
        int[] parts = new int[n];
        int sum = 0;
        for (int i = 0; i < n - 1; i++) {
            parts[i] = rand.nextInt(100);
            sum += parts[i];
        }
        parts[n - 1] = secret - sum;

        System.out.println("Секрет разделён на части:");
        for (int i = 0; i < n; i++) {
            System.out.println("Часть " + (i+1) + ": " + parts[i]);
        }
    }

    public static void combineSecret() {
        Scanner sc = new Scanner(System.in);
        System.out.print("Введите количество частей: ");
        int n = sc.nextInt();
        int secret = 0;
        for (int i = 0; i < n; i++) {
            System.out.print("Часть " + (i+1) + ": ");
            secret += sc.nextInt();
        }
        System.out.println("Восстановленный секрет: " + secret);
    }
}
```

### SSSSen.java

```java
import java.util.*;

public class SSSSen {
    private static final int PRIME = 257; // простое > 255

    private static int modPow(int base, int exp, int mod) {
        int res = 1;
        base %= mod;
        while (exp > 0) {
            if ((exp & 1) == 1) res = (res * base) % mod;
            base = (base * base) % mod;
            exp >>= 1;
        }
        return res;
    }

    public static List<int[]> shamirSplit(int secret, int n, int k) {
        Random rand = new Random();
        int[] coeffs = new int[k];
        coeffs[0] = secret;
        for (int i = 1; i < k; i++) {
            coeffs[i] = rand.nextInt(PRIME);
        }

        List<int[]> shares = new ArrayList<>();
        for (int x = 1; x <= n; x++) {
            int y = 0;
            int xp = 1;
            for (int c : coeffs) {
                y = (y + c * xp) % PRIME;
                xp = (xp * x) % PRIME;
            }
            shares.add(new int[]{x, y});
        }
        return shares;
    }

    public static int shamirCombine(int[][] shares) {
        int k = shares.length;
        int secret = 0;
        for (int i = 0; i < k; i++) {
            int xi = shares[i][0];
            int yi = shares[i][1];

            int num = 1;
            int den = 1;
            for (int j = 0; j < k; j++) {
                if (i == j) continue;
                int xj = shares[j][0];
                num = (num * (PRIME - xj)) % PRIME;
                den = (den * (xi - xj + PRIME)) % PRIME;
            }

            // обратный элемент через modPow
            int inv = modPow(den, PRIME - 2, PRIME);
            int li = (num * inv) % PRIME;
            secret = (secret + yi * li) % PRIME;
        }
        return secret;
    }
}
```
